name: Update News JSON

on:
  workflow_dispatch:
  schedule:
    - cron: "0 */6 * * *"   # every 6 hours (UTC)

permissions:
  contents: write

jobs:
  build-news:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Fetch RSS and build news.json
        run: |
          mkdir -p assets

          python - << 'PY'
          import json, time, datetime
          import urllib.request
          import xml.etree.ElementTree as ET

          def fetch(url, timeout=30):
            req = urllib.request.Request(url, headers={"User-Agent":"Mozilla/5.0"})
            with urllib.request.urlopen(req, timeout=timeout) as r:
              return r.read()

          def parse_rss(xml_bytes, category):
            root = ET.fromstring(xml_bytes)
            channel = root.find("channel")
            items = channel.findall("item") if channel is not None else []
            out = []
            for it in items[:12]:
              title = (it.findtext("title") or "").strip()
              link = (it.findtext("link") or "").strip()
              pub  = (it.findtext("pubDate") or "").strip()
              source = ""
              src_el = it.find("{http://search.yahoo.com/mrss/}credit")
              if src_el is not None and src_el.text:
                source = src_el.text.strip()

              if not source:
                # Google News often puts source in title like "Headline - Source"
                if " - " in title:
                  parts = title.rsplit(" - ", 1)
                  title = parts[0].strip()
                  source = parts[1].strip()

              # Description is HTML; keep short text only
              desc = (it.findtext("description") or "").strip()
              # quick strip tags
              import re
              desc_txt = re.sub("<[^>]*>", "", desc).strip()
              if len(desc_txt) > 220:
                desc_txt = desc_txt[:220].rstrip() + "…"

              out.append({
                "category": category,
                "title": title,
                "source": source or "Google News",
                "link": link,
                "published": pub,
                "summary": desc_txt
              })
            return out

          # --- Queries tuned for Orchard Diary (India) ---
          queries = {
            "pest": "apple orchard pest disease San Jose scale aphid thrips borer scab mildew himachal",
            "govt": "horticulture department advisory scheme subsidy notification apple himachal",
            "weather": "frost alert hailstorm rainfall warning himachal apple orchard weather advisory"
          }

          def google_rss(query):
            import urllib.parse
            q = urllib.parse.quote(query)
            return f"https://news.google.com/rss/search?q={q}&hl=en-IN&gl=IN&ceid=IN:en"

          all_items = []
          for cat, q in queries.items():
            url = google_rss(q)
            xml = fetch(url)
            all_items.extend(parse_rss(xml, cat))

          data = {
            "generated_at_utc": datetime.datetime.utcnow().isoformat() + "Z",
            "items": all_items
          }

          with open("assets/news.json", "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

          print("Wrote assets/news.json with", len(all_items), "items")
          PY

      - name: Commit and push if changed
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add assets/news.json
          git diff --cached --quiet || git commit -m "chore: update news.json"
          git push
